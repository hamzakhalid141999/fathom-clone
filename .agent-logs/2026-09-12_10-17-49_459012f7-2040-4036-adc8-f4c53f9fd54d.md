## TypeScript interfaces

```ts
export type MeetingId = string;

export interface Participant {
  id: string;
  name: string;
  email?: string;
  role?: string;          // e.g. "Host", "AE", "Customer"
  avatarColor?: string;   // for initials chip
}

export interface TranscriptSegment {
  id: string;
  speakerId: string;      // → Participant.id
  speakerName: string;    // denormalized for render speed
  startMs: number;        // offset from meeting start
  endMs: number;
  text: string;
}

export type ActionItemStatus = "open" | "done" | "cancelled";

export interface ActionItem {
  id: string;
  text: string;
  assigneeId?: string;
  assigneeName?: string;
  dueDate?: string;       // ISO date
  status: ActionItemStatus;
  sourceTimestampMs?: number; // jump-to in transcript/player
}

export interface Clip {
  id: string;
  title: string;
  startMs: number;
  endMs: number;
  thumbnailLabel?: string; // stub — no real video frames
  createdBy?: string;
}

export interface MeetingSummary {
  headline: string;       // one-liner
  bullets: string[];      // key points
  decisions?: string[];
  nextSteps?: string[];
}

export interface Meeting {
  id: MeetingId;
  title: string;
  startedAt: string;      // ISO datetime
  durationMs: number;
  platform: "zoom" | "meet" | "teams" | "other";
  participants: Participant[];
  summary: MeetingSummary;
  transcript: TranscriptSegment[];
  actionItems: ActionItem[];
  clips: Clip[];
  tags?: string[];
  recordingStubUrl?: string; // fake; player is UI-only
}
```
